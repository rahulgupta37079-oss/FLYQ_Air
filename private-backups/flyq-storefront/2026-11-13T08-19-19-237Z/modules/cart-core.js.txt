export const catalog=Object.freeze({remote:{name:'Remote',price:150000},holder:{name:'Remote with holder',price:180000},motors:{name:'Drone motors set',price:99900},propellers:{name:'Propeller set',price:59900},battery:{name:'Extra battery',price:69900}});
export function cleanCart(value){const out={};if(!value||typeof value!=='object'||Array.isArray(value))return out;for(const id of Object.keys(catalog)){if(Number.isInteger(value[id])&&value[id]>0)out[id]=Math.min(10,value[id]);}return out;}
export function totals(value){const cart=cleanCart(value);const subtotal=Object.entries(cart).reduce((sum,[id,q])=>sum+catalog[id].price*q,0);const discount=Math.round(subtotal*8/100);return {subtotal,discount,total:subtotal-discount};}
export const money=n=>'₹'+(n/100).toLocaleString('en-IN',{minimumFractionDigits:2,maximumFractionDigits:2});
export function orderText(value){const cart=cleanCart(value),t=totals(cart);return 'FLYQ KIT V1 order request\n'+Object.entries(cart).map(([id,q])=>catalog[id].name+' × '+q+' — '+money(catalog[id].price*q)).join('\n')+'\nSubtotal: '+money(t.subtotal)+'\n8% offer: −'+money(t.discount)+'\nParts total: '+money(t.total)+'\nPlease confirm availability, shipping and payment arrangements.';}
