import {catalog,cleanCart,totals,money,orderText} from './cart-core.js';
let cart={};try{cart=cleanCart(JSON.parse(localStorage.getItem('flyq-kit-v1-cart')||'{}'));}catch{}
function render(){const root=document.getElementById('cart-items');root.replaceChildren();for(const [id,q] of Object.entries(cart)){const row=document.createElement('div');row.className='cart-line';const label=document.createElement('span');label.textContent=catalog[id].name+' · '+money(catalog[id].price*q);row.append(label);for(const [text,change,aria] of [['−',-1,'Decrease'],['+',1,'Increase']]){const b=document.createElement('button');b.type='button';b.textContent=text;b.setAttribute('aria-label',aria+' '+catalog[id].name+' quantity');b.disabled=change===1&&q===10;b.onclick=()=>{cart[id]+=change;cart=cleanCart(cart);save();};row.append(b);if(change===-1){const n=document.createElement('strong');n.textContent=String(q);row.append(n);}}const remove=document.createElement('button');remove.textContent='Remove';remove.setAttribute('aria-label','Remove '+catalog[id].name);remove.onclick=()=>{delete cart[id];save();};row.append(remove);root.append(row);}const t=totals(cart);document.getElementById('cart-summary').textContent=t.subtotal?'Subtotal '+money(t.subtotal)+' · You save '+money(t.discount)+' (8%) · Parts total '+money(t.total):'Your cart is empty.';const order=document.getElementById('cart-order');order.hidden=!t.subtotal;if(t.subtotal)order.href='https://wa.me/919137361474?text='+encodeURIComponent(orderText(cart));else order.removeAttribute('href');}
function save(){try{localStorage.setItem('flyq-kit-v1-cart',JSON.stringify(cart));}catch{}render();}
for(const b of document.querySelectorAll('[data-add]'))b.addEventListener('click',()=>{const id=b.dataset.add;if(!Object.hasOwn(catalog,id))return;cart[id]=Math.min(10,(cart[id]||0)+1);save();document.getElementById('kit-cart').scrollIntoView({behavior:matchMedia('(prefers-reduced-motion: reduce)').matches?'instant':'smooth'});});
for(const img of document.querySelectorAll('.kit-photo img'))img.addEventListener('error',()=>{if(!img.dataset.fallback){img.dataset.fallback='1';img.src=img.src.replace('.jpeg','-original.jpeg');}});
render();
